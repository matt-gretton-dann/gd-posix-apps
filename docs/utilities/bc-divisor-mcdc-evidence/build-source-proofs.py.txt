#!/usr/bin/env python3
"""Retain Git objects needed to authenticate the archived source subsets."""
import base64,hashlib,json,subprocess
from pathlib import Path
ROOT=Path('/private/tmp/gd-posix-divisor-mcdc-evidence/docs/utilities/bc-divisor-mcdc-evidence')
REPO=Path('/Users/mgrettondann/repos/gd-posix-apps')
SCOPES=('src','libgdsup','cmake','python-lib','CMakeLists.txt')
def git(*args):return subprocess.check_output(['git',*args],cwd=str(REPO))
def entries(raw):
 at=0
 while at<len(raw):
  space=raw.index(b' ',at);end=raw.index(b'\0',space);oid=raw[end+1:end+21].hex()
  yield raw[at:space].decode(),raw[space+1:end].decode(),oid
  at=end+21
 assert at==len(raw)
def main():
 inventory=json.loads((ROOT/'inventory.json').read_text());directory=ROOT/'source-proofs';directory.mkdir(exist_ok=True)
 proofs={}
 for revision in sorted({e['source_revision'] for e in inventory['collections']}):
  commit=git('cat-file','commit',revision);root=commit.splitlines()[0].split()[1].decode();trees={}
  def walk(oid):
   if oid in trees:return
   raw=git('cat-file','tree',oid);trees[oid]=base64.b64encode(raw).decode()
   for mode,name,child in entries(raw):
    if mode=='40000':walk(child)
  raw=git('cat-file','tree',root);trees[root]=base64.b64encode(raw).decode()
  for mode,name,oid in entries(raw):
   if name in SCOPES and mode=='40000':walk(oid)
  proof=dict(revision=revision,scopes=SCOPES,commit=base64.b64encode(commit).decode(),trees=trees)
  path=directory/(revision+'.json');path.write_text(json.dumps(proof,indent=2)+'\n')
  proofs[revision]=dict(path=str(path.relative_to(ROOT)),sha256=hashlib.sha256(path.read_bytes()).hexdigest())
 inventory['source_proofs']=proofs
 (ROOT/'inventory.json').write_text(json.dumps(inventory,indent=2)+'\n')
 (ROOT/'build-source-proofs.py.txt').write_bytes(Path(__file__).read_bytes())
 print('Retained commit/root/scoped-tree proofs for',len(proofs),'immutable source revisions')
if __name__=='__main__':main()
