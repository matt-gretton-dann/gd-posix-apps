"""Collect logical allocations and native RSS before the Toom elapsed grid."""
import hashlib,importlib.util,json,os,shutil,subprocess,time
from pathlib import Path
scripts=Path('/private/tmp/gd-posix-toom3-prototype/src/bc/scripts')
def load(name):
 spec=importlib.util.spec_from_file_location(name,scripts/(name+'.py'));m=importlib.util.module_from_spec(spec);spec.loader.exec_module(m);return m
bench=load('bench-bc');profile=load('profile-bc')
preparation=Path('/private/tmp/bc-204-toom-grid-preparation');prepared=json.loads((preparation/'complete.json').read_text());assert prepared['checked']==888
root=Path('/private/tmp/bc-204-toom-grid-diagnostics');root.mkdir(exist_ok=False)
source=Path('/private/tmp/gd-posix-toom3-prototype');head=subprocess.check_output(['git','rev-parse','HEAD'],cwd=source).decode().strip()
working=subprocess.check_output(['git','status','--porcelain'],cwd=source).decode();assert not working
runtime_changes=subprocess.check_output(['git','diff','--name-only',prepared['source_revision'],head,'--','src','libgdsup','cmake','CMakeLists.txt',':(exclude)src/bc/test-*'],cwd=source).decode();assert not runtime_changes
profiles={}
for threshold in (0,128,512,2048):
 directory=source/('build-profile' if threshold==512 else 'build-profile-'+str(threshold));binary=directory/'src/bc/bc';assert binary.is_file()
 profiles[threshold]=dict(argv=[str(binary)],binary_sha256=bench.digest_file(binary),cache_sha256=bench.digest_file(directory/'CMakeCache.txt'))
 shutil.copyfile(directory/'CMakeCache.txt',root/('profile-'+str(threshold)+'-CMakeCache.txt'))
 shutil.copyfile(directory/'build.ninja',root/('profile-'+str(threshold)+'-build.ninja.txt'))
roles=bench.implementations(preparation/'training-paired.json',True)
manifest=dict(source_revision=prepared['source_revision'],collection_head=head,runtime_changes=runtime_changes,working_changes=working,profiles=profiles,roles=roles,collector_sha256=bench.digest_file(__file__),purpose='Separate allocation/RSS diagnostics; no elapsed speed claim. Fresh evaluation diagnostics are retained without selecting thresholds from their outcomes.',logical=[],resources=[],cases=[])
started=time.monotonic()
try:
 for preset in ('full','holdout'):
  cases,replay=bench.replay_corpus(preparation/preset,'arithmetic')
  for case in [c for c in cases if c['name'].startswith('toom_grid_')]:
   if time.monotonic()-started>7200:raise RuntimeError('diagnostic budget exhausted')
   name=preset+'-'+case['name'];payload=case['program'].encode();oracle=bench.oracle_payload(case)
   (root/(name+'.bc')).write_bytes(payload);(root/(name+'.expected')).write_bytes(oracle)
   manifest['cases'].append(dict(name=name,program_sha256=hashlib.sha256(payload).hexdigest(),oracle_sha256=hashlib.sha256(oracle).hexdigest(),parameters=case['parameters']))
   for mode in ('single','batched'):
    selected=dict(case)
    if mode=='single':
     count=case['parameters']['iterations'];selected['program']=case['program'].replace('i<'+str(count)+';','i<1;')
    for threshold,entry in profiles.items():
     destination=root/(name+'-'+mode+'-'+str(threshold)+'.profile.json')
     env=bench.environment();env['GD_BC_PROFILE_OUTPUT']=str(destination)
     child=profile.run_group(entry['argv'],selected['program'].encode(),env,120)
     assert child.returncode==0 and not child.stderr,(name,mode,threshold,child.stderr)
     bench.check_output(selected,bench.canonical_output(child.stdout),'profile/'+name)
     report=json.loads(destination.read_text());assert report['instrumented'] is True and report['toom3_threshold']==threshold and report['multiply_split_point']==126
     assert report['total']['live_bytes']==0 and report['total']['allocation_calls']==report['total']['deallocation_calls']
     manifest['logical'].append(dict(name=name,mode=mode,threshold=threshold,checked=True,report=destination.name,sha256=bench.digest_file(destination),program_sha256=hashlib.sha256(selected['program'].encode()).hexdigest()))
   for trial in range(3):
    ordered=roles if trial==0 else list(reversed(roles)) if trial==1 else roles[1:]+roles[:1]
    for role in ordered:
     report=root/(name+'-'+role['name']+'-'+str(trial)+'.rss.txt')
     env=bench.environment();env.pop('GD_BC_PROFILE_OUTPUT',None)
     child=profile.run_group(['/usr/bin/time','-l','-o',str(report)]+role['argv'],payload,env,120)
     assert child.returncode==0 and not child.stderr,(name,role['name'],child.stderr)
     bench.check_output(case,bench.canonical_output(child.stdout),'rss/'+name)
     fields=profile.rss_fields('Darwin',report.read_text());assert fields['status']=='collected'
     manifest['resources'].append(dict(name=name,role=role['name'],trial=trial,checked=True,report=report.name,report_sha256=bench.digest_file(report),resources=fields))
   bench.write_json(root/'manifest.json',manifest);print(name,'diagnostics checked',flush=True)
 manifest['status']='all separate diagnostic children checked; no timing acceptance claim'
except BaseException as error:
 manifest['failure']=str(error);raise
finally:bench.write_json(root/'manifest.json',manifest)
