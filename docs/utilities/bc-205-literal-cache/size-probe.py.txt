import json,shlex,subprocess
from pathlib import Path
root=Path('/private/tmp/gd-bc-205-cache-sizes');root.mkdir(exist_ok=False)
source=root/'probe.cc';source.write_text('#include <cstdio>\n#include "bc.hh"\nint main(){std::printf("{\\"instruction_bytes\\":%zu,\\"number_bytes\\":%zu}\\n",sizeof(GD::Bc::Instruction),sizeof(GD::Bc::Number));}\n')
rows=[]
for role,directory in [('parent','/private/tmp/gd-bc-202-integrated-release'),('candidate','/private/tmp/gd-bc-205-cache-release')]:
 directory=Path(directory);entries=json.loads((directory/'compile_commands.json').read_text());entry=next(e for e in entries if e['file'].endswith('/src/bc/bc.cc'))
 original=entry.get('arguments') or shlex.split(entry['command']);args=[];skip=False
 for arg in original:
  if skip:skip=False;continue
  if arg in ('-o','-MT','-MF','-MQ'):skip=True;continue
  if arg in ('-c','-MD','-MMD') or arg==entry['file']:continue
  args.append(arg)
 binary=root/role;command=args+[str(source),'-o',str(binary)]
 subprocess.run(command,cwd=entry['directory'],check=True,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
 row=json.loads(subprocess.check_output([str(binary)],text=True));row.update(role=role,command=command,source_revision=json.loads((directory/'implementations.json').read_text())['implementations'][0]['metadata']['source_revision']);rows.append(row)
 print(role,row['instruction_bytes'],row['number_bytes'])
(root/'sizes.json').write_text(json.dumps(rows,indent=2)+'\n')
