from pathlib import Path
import hashlib,json,sys,zipfile
for os in sys.argv[1:]:
 p=Path('/private/tmp/bc-210-parser-native-'+os+'.zip');z=zipfile.ZipFile(p);names=z.namelist();identities={}
 for role,expected in [('base','ac5eee3cbc494ac3c5f63ce2bd2e9b71cc74271f'),('candidate','9c65f912ac9ff5c32503f8d25d404e439164297f')]:
  entries=json.loads(z.read(role+'/implementations.json'))['implementations'];assert len(entries)==1
  identities[role]=entries[0]['metadata']['source_revision'];assert identities[role]==expected
  assert entries[0]['metadata']['instrumented'] is False
 rows=[]
 for n in names:
  if n.endswith('/samples.jsonl'):
   part=[json.loads(s) for s in z.read(n).decode().splitlines()];assert part and all(x['checked'] for x in part);rows+=part
 assert len(rows)>=1700
 summary=z.read('summary.md').decode();confirmed=[line for line in summary.splitlines() if 'confirmed regression' in line.lower().replace('-', ' ')];selected=[line for line in summary.splitlines() if '| paired |' in line and any('`'+n+'`' in line for n in ['loop_1000000','calls_200000','array_200000','frontend_definitions_2000','streaming_statements_20000','temporary_chain_1000d_x50000'])]
 audit={'os':os,'checked':len(rows),'identities':identities,'confirmed':confirmed,'selected':selected,'sha256':hashlib.sha256(p.read_bytes()).hexdigest()};p.with_suffix('.audit.json').write_text(json.dumps(audit,indent=2)+'\n');print(json.dumps(audit))
